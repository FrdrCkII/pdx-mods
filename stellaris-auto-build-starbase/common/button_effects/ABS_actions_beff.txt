ABS_build_starbases = {
    potential = { stri_ABS_mod = yes }
    allow = { is_ai = no }
    effect = {
        custom_tooltip = ABS_build_starbases_tt
        set_country_flag = flag_ABS_action
        save_event_target_as = ABSCountry
        every_system = {
            limit = {
                stri_ABS_can_build_starbase = yes
                is_surveyed = { who = event_target:ABSCountry status = yes }
            }
            save_event_target_as = ABSSystem
            seff_ABS_build_starbase = yes
            if = {
                limit = { event_target:ABSCountry = { has_country_flag = flag_ABS_build_starbases_with_stations } }
                seff_ABS_build_station = yes
            }
        }
        if = {
            limit = { has_country_flag = flag_ABS_build_starbases_subject }
            every_subject = {
                save_event_target_as = ABSSubject
                every_system = {
                    limit = {
                        stri_ABS_can_build_starbase = yes
                        is_surveyed = { who = event_target:ABSSubject status = yes }
                    }
                    save_event_target_as = ABSSystem
                    seff_ABS_build_starbase = yes
                    if = {
                        limit = { event_target:ABSCountry = { has_country_flag = flag_ABS_build_starbases_with_stations } }
                        seff_ABS_build_station = yes
                    }
                }
            }
        }
        remove_country_flag = flag_ABS_action
    }
}
