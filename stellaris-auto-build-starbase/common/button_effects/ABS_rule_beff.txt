    potential = { stri_ABS_mod = yes }
    allow = { is_ai = no }

        custom_tooltip = ABS_auto_build_starbases_tt
        if = {
            limit = { has_country_flag = flag_ABS_auto_build_starbases }
            remove_country_flag = flag_ABS_auto_build_starbases

    }
