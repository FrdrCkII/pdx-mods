country_event = {
    id = ABS.1
    is_triggered_only = yes
    hide_window = yes

    pre_triggers = {
        is_ai = no
    }

    trigger = {
        from = {
            NOT = { exists = starbase }
        }
        root = {
            stri_ABS_is_auto_build_on = yes
        }
    }

    immediate = {
        if = {
            limit = { stri_ABS_is_auto_build_on_subject = yes }
            from = { save_event_target_as = ABSSystem }
            root = { save_event_target_as = ABSSubject }
            root.overlord = { save_event_target_as = ABSCountry }
        }else = {
            from = { save_event_target_as = ABSSystem }
            root = { save_event_target_as = ABSCountry }
        }
        seff_ABS_auto_build = yes
    }
}

starbase_event = {
    id = ABS.2
    hide_window = yes
    is_triggered_only = yes
    trigger = {
        from = {
            is_ai = no
            has_country_flag = flag_ABS_auto_build_starbases_stations
        }
    }
    immediate = {
        this.solar_system = { save_event_target_as = ABSSystem }
        from = { save_event_target_as = ABSCountry }
        seff_ABS_build_station = yes
    }
}

    desc = {

    immediate = {
        save_event_target_as = ABSCountry
    }

    option = {

    }
    option = {
        name = ABS.esc
    }

    after = {
        remove_country_flag = flag_ABS_is_subject
    }
